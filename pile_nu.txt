on journal texte
  put texte & return after card field "R" of card "Atelier"
end journal
on titre t
  if card field "R" of card "Atelier" is not empty then journal empty
  journal "== " & t
end titre
on egal quoi, obtenu, attendu
  global gOK, gKO
  if obtenu is attendu then
    journal "  ok     " & quoi
    add 1 to gOK
  else
    journal "  ECHEC  " & quoi
    journal "         attendu [" & attendu & "]"
    journal "         obtenu  [" & obtenu & "]"
    add 1 to gKO
  end if
end egal
on verifie quoi, cond
  global gOK, gKO
  if cond then
    journal "  ok     " & quoi
    add 1 to gOK
  else
    journal "  ECHEC  " & quoi
    add 1 to gKO
  end if
end verifie
on note quoi, valeur
  global gNotes
  journal "  ?      " & quoi & " : [" & valeur & "]"
  add 1 to gNotes
end note
on bilan
  global gOK, gKO, gNotes
  journal empty
  journal "=============================="
  if gKO is 0 then
    journal gOK & " reussites, aucun echec, " & gNotes & " a mesurer."
    put gOK & " reussites, aucun echec, " & gNotes & " a mesurer."
  else
    journal gOK & " reussites, " & gKO & " ECHECS, " & gNotes & " a mesurer."
    put gOK & " reussites, " & gKO & " ECHECS — voir le champ R."
  end if
end bilan

on relitClavier
  global gVus, gClavier
  if gVus is empty then
    put "on" into gClavier
    answer "Journal du clavier ALLUME. Tapez une lettre, une fleche, tabulation, entree, puis recliquez ici. (Les touches ne feront plus rien d'autre tant qu'il est allume.)"
  else
    journal empty
    journal "== les touches tapees, dans l'ordre =="
    journal "  " & gVus
    put empty into gVus
    put "off" into gClavier
  end if
end relitClavier

function terrainBon
  global gTerrain, gTerrainDit
  put "bon" into gTerrain
  put empty into gTerrainDit
  if the number of cards < 4 then
    manque "il faut QUATRE cartes au moins ; il y en a " & the number of cards
  end if
  exigeCarte "Atelier"
  exigeCarte "Deux"
  exigeCarte "Trois"
  exigeCarte "Quatre"
  if there is not a card "Atelier" then
    manque "la premiere carte doit se nommer « Atelier »"
  else
    go to card "Atelier"
    if the number of card fields < 3 then
      manque "l'Atelier doit porter TROIS champs de carte, A puis B puis R ; il en a " & the number of card fields
    else
      if the short name of card field 1 is not "A" then
        manque "card field 1 doit se nommer « A » ; il se nomme « " & the short name of card field 1 & " » — L'ORDRE DE CREATION EST LE NUMERO, et les sections 3, 4 et 7 le verifient"
      end if
      if the short name of card field 2 is not "B" then
        manque "card field 2 doit se nommer « B » ; il se nomme « " & the short name of card field 2 & " »"
      end if
      if the short name of card field 3 is not "R" then
        manque "card field 3 doit se nommer « R » ; il se nomme « " & the short name of card field 3 & " »"
      end if
      if the dontSearch of card field "R" is not true then
        manque "le champ R doit avoir dontSearch a VRAI : il porte le nom de chaque essai, donc tous les motifs cherches, et la section 1 se trouverait elle-meme"
      end if
    end if
    if the number of bkgnd fields < 1 then
      manque "il faut un champ de FOND nomme « T », present sur les quatre cartes"
    else
      if the short name of bkgnd field 1 is not "T" then
        manque "bkgnd field 1 doit se nommer « T » ; il se nomme « " & the short name of bkgnd field 1 & " »"
      end if
    end if
  end if
  return gTerrain is "bon"
end terrainBon
on exigeCarte quel
  if there is not a card quel then
    manque "pas de carte nommee « " & quel & " »"
  end if
end exigeCarte
on manque quoi
  global gTerrain, gTerrainDit
  put "casse" into gTerrain
  if gTerrainDit is empty then put quoi into gTerrainDit
  else put gTerrainDit & return & return & quoi into gTerrainDit
end manque
