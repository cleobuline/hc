-- Compare et note. Le seul endroit qui écrit le rapport.
on verif quoi, attendu, obtenu
  global gEssais, gRates, gSortie
  add 1 to gEssais
  if obtenu is attendu then
    put "ok   " & quoi & " = " & aplat(obtenu) & return after gSortie
  else
    put "RATE " & quoi & " : attendu <" & aplat(attendu) & "> obtenu <" & aplat(obtenu) & ">" & return after gSortie
    put quoi & return after gRates
  end if
end verif

-- Une valeur qui contient des retours tiendrait sur plusieurs lignes du
-- champ et casserait le « une ligne par essai » : on les remplace.
function aplat t
  repeat while offset(return, t) > 0
    put "/" into char offset(return, t) of t
  end repeat
  return t
end aplat

-- -------------------------------------------------------------- controle
on testControle
  put 0 into s
  repeat with i = 1 to 5
    add i to s
  end repeat
  verif "repeat with", 15, s

  put 0 into s
  repeat with i = 5 down to 1
    add 1 to s
  end repeat
  verif "repeat down to", 5, s

  put 0 into s
  repeat 3 times
    add 1 to s
  end repeat
  verif "repeat times", 3, s

  put 0 into s
  repeat until s >= 4
    add 1 to s
  end repeat
  verif "repeat until", 4, s

  put 0 into s
  repeat while s < 3
    add 1 to s
  end repeat
  verif "repeat while", 3, s

  put 0 into s
  repeat with i = 1 to 10
    if i > 3 then exit repeat
    add 1 to s
  end repeat
  verif "exit repeat", 3, s

  put 0 into s
  repeat with i = 1 to 5
    if i mod 2 = 0 then next repeat
    add 1 to s
  end repeat
  verif "next repeat", 3, s

  if 1 = 1 then
    put "oui" into z
  else
    put "non" into z
  end if
  verif "if then else", "oui", z

  put "" into z
  if 1 = 2 then put "a" into z
  else put "b" into z
  verif "if une ligne", "b", z
end testControle

-- ------------------------------------------------------------ navigation
-- Non destructif : on empile la carte de départ et on y revient.
on testNavigation
  -- « lock messages » : sans lui, chaque changement de carte declenche les
  -- openCard/closeCard DE TA PILE, dont les gestionnaires polluent le bilan
  -- (et peuvent echouer sur une carte qu ils ne connaissent pas).
  lock messages
  push card
  put the short name of this card into depart
  put the number of this card into rangDepart

  go to next card
  verif "go next puis retour", true, (the number of this card) is not empty
  go to previous card
  verif "go previous ramene", rangDepart, the number of this card

  go to first card
  verif "go first", 1, the number of this card
  go to last card
  verif "go last", the number of cards, the number of this card

  go to card rangDepart
  verif "go card par rang", depart, the short name of this card

  pop card
  verif "pop card", depart, the short name of this card

  go to card "carteQuiNexistePas"
  verif "go vers l inexistant remplit le result", true, the result is not empty
  go to card rangDepart
  unlock messages
end testNavigation
