--  TORTURE 2 — LE SCRIPT DE LA PILE (Objects > Stack Info... > Script)
--
--  Il porte deux choses :
--
--   1. L'ECHAFAUDAGE du rapport — journal, titre, egal, note, bilan.
--      Il est ICI et non sur le bouton pour que n'importe quel bouton de
--      n'importe quelle carte puisse s'en servir.
--
--   2. LES DIX GESTIONNAIRES DU CLAVIER. Les messages du clavier sont
--      livres a la carte courante, d'ou ils remontent au fond puis a la
--      pile : un bouton ne les voit jamais. C'est donc ici qu'ils
--      doivent etre ecrits, et nulle part ailleurs.
--
--  UNE FOIS LA PILE OUVERTE, les touches marchent pour de vrai : tapez
--  une lettre, une fleche, tabulation, entree — le journal gVus les
--  note, et le bouton « clavier » le relit.
-- ════════════════════════════════════════════════════════════════════


-- ---- l'echafaudage du rapport --------------------------------------

-- Une ligne de plus dans le rapport, ECRITE TOUT DE SUITE.
--
-- Le rapport ne passe pas par une variable recopiee a la fin : si un
-- essai leve une erreur, tout ce qui precede est deja dans le champ et
-- l'on sait jusqu'ou le script est alle. Un rapport garde en memoire
-- serait perdu par la premiere faute.
--
-- « texte & return after » et non « return & texte after » : le second
-- met le saut de ligne DEVANT, ce qui laisse une premiere ligne vide et
-- decale tout le rapport d'un cran.
on journal texte
  put texte & return after card field "R" of card "Atelier"
end journal

-- Une ligne vide AVANT chaque titre, sauf le premier.
on titre t
  if card field "R" of card "Atelier" is not empty then journal empty
  journal "== " & t
end titre

-- UN ESSAI QUI A UNE REPONSE ATTENDUE, et qui dit les DEUX valeurs
-- quand il echoue. « ECHEC find whole » n'apprend rien ; « attendu
-- [Not found] obtenu [not found] » nomme la cause a lui seul.
on egal quoi, obtenu, attendu
  global gOK, gKO
  if obtenu is attendu then
    journal "  ok     " & quoi
    add 1 to gOK
  else
    journal "  ECHEC  " & quoi
    journal "         attendu [" & attendu & "]"
    journal "         obtenu  [" & obtenu & "]"
    add 1 to gKO
  end if
end egal

-- La forme courte, pour une condition deja ecrite dans l'appel.
on verifie quoi, cond
  global gOK, gKO
  if cond then
    journal "  ok     " & quoi
    add 1 to gOK
  else
    journal "  ECHEC  " & quoi
    add 1 to gKO
  end if
end verifie

-- CE QUI N'EST PAS MESURE NE SE COMPTE PAS.
--
-- « note » affiche une valeur SANS verdict : personne n'a encore
-- regarde ce qu'HyperCard en fait. Une ligne « note » n'est ni une
-- reussite ni un echec — c'est une question posee, a jouer dans
-- Basilisk a cote pour la trancher. Les compter en reussites serait
-- inscrire une supposition comme un fait.
on note quoi, valeur
  global gNotes
  journal "  ?      " & quoi & " : [" & valeur & "]"
  add 1 to gNotes
end note

-- Le detail est deja dans le champ, ligne a ligne ; la boite de message
-- ne recoit que le resume — « put » vers elle passe par un tampon de
-- 1024 octets et couperait un long rapport en silence.
on bilan
  global gOK, gKO, gNotes
  journal empty
  journal "=============================="
  if gKO is 0 then
    journal gOK & " reussites, aucun echec, " & gNotes & " a mesurer."
    put gOK & " reussites, aucun echec, " & gNotes & " a mesurer."
  else
    journal gOK & " reussites, " & gKO & " ECHECS, " & gNotes & " a mesurer."
    put gOK & " reussites, " & gKO & " ECHECS — voir le champ R."
  end if
end bilan

on razCompteurs
  global gOK, gKO, gNotes
  put 0 into gOK
  put 0 into gKO
  put 0 into gNotes
  put empty into card field "R" of card "Atelier"
end razCompteurs

-- LES ERREURS N'OUVRENT PLUS DE DIALOGUE : ELLES S'ECRIVENT DANS LE RAPPORT.
--
-- Le banc PROVOQUE des erreurs — c'est ainsi qu'on verifie que HC s'en
-- plaint au lieu de rendre sa propre phrase — et chacune ouvrait une
-- alerte modale qu'il fallait cliquer. Deux prix a cela : la course
-- s'arretait pour de bon si l'on cliquait de travers, et l'utilisatrice
-- n'avait plus qu'a commenter la ligne fautive pour en finir — ce qui
-- fait echouer l'essai qui la surveille, et a juste titre, puisque plus
-- rien ne leve.
--
-- « set the lockErrorDialogs to true » detourne les erreurs vers ce
-- gestionnaire. MESURE : le detournement marche, et l'erreur arrete
-- toujours le gestionnaire ou elle tombe — c'est ce qu'on veut, et c'est
-- ce que les essais isoles mesurent.
--
-- ATTENTION AU MOMENT : les erreurs sont accumulees et videes a la sortie
-- du gestionnaire LE PLUS EXTERIEUR. Ce bloc s'ecrit donc APRES le bilan,
-- tout a la fin du rapport, et « gErr » ne peut pas se relire dans le
-- meme clic. Mon premier essai remettait le reglage a false avant la
-- sortie : le dialogue revenait, et j'ai cru une seconde que
-- lockErrorDialogs ne marchait pas.
on errorDialog quoi
  journal empty
  journal "== les erreurs levees, et elles sont VOULUES =="
  repeat with i = 1 to the number of lines of quoi
    journal "  " & line i of quoi
  end repeat
  -- On rend la main aux dialogues : le reglage est global, et une pile qui
  -- avale les erreurs suivantes en silence serait pire que le dialogue.
  set the lockErrorDialogs to false
end errorDialog


-- ---- les dix gestionnaires du clavier -------------------------------
--
-- Chacun note son nom et son argument dans gVus, et RIEN D'AUTRE : le
-- bouton « clavier » relit ce journal. Un gestionnaire qui ecrirait
-- dans le champ du rapport melangerait la mesure et son compte rendu.

on keyDown k
  vu "keyDown " & k
end keyDown

on commandKeyDown k
  vu "commandKeyDown " & k
end commandKeyDown

on controlKey n
  vu "controlKey " & n
end controlKey

on functionKey n
  vu "functionKey " & n
end functionKey

on arrowKey d
  vu "arrowKey " & d
end arrowKey

on enterKey
  vu "enterKey"
end enterKey

on returnKey
  vu "returnKey"
end returnKey

on tabKey
  vu "tabKey"
end tabKey

on enterInField
  vu "enterInField"
end enterInField

on returnInField
  vu "returnInField"
end returnInField

-- Le journal des touches, dans une globale pour qu'il survive d'un
-- message a l'autre.
on vu quoi
  global gVus
  if gVus is empty then put quoi into gVus
  else put gVus & "," & quoi into gVus
end vu


-- ---- les trois boutons d'appoint -----------------------------------
--
-- Ce que le script ne peut PAS mesurer tout seul : une touche tapee
-- pour de vrai, et une selection posee a la main. Une commande part du
-- script, une touche part de l'interface, et ce sont deux chemins
-- differents dans HC — c'est justement celui de l'interface qu'aucun
-- harnais ne peut emprunter.

-- Bouton « clavier » : relit le journal des touches, puis le vide.
on relitClavier
  global gVus
  if gVus is empty then
    answer "Rien de tape. Tapez une lettre, une fleche, tabulation, entree, puis recliquez."
  else
    journal empty
    journal "== les touches tapees, dans l'ordre =="
    journal "  " & gVus
    put empty into gVus
  end if
end relitClavier

-- Bouton « selection » : lit les quatre proprietes de la selection SANS
-- rien ecrire entre elles.
--
-- L'ORDRE DE CES QUATRE LIGNES COMPTE, et c'est une lecon payee cher :
-- un banc qui ecrivait dans un champ entre chaque lecture pouvait
-- defaire la selection qu'il mesurait, et l'on a cru trois fois a un
-- defaut de HC qui n'existait pas. Les quatre valeurs sont donc prises
-- d'affilee dans des variables, et le rapport s'ecrit APRES.
on relitSelection
  put the selectedChunk into ch
  put the selectedField into fl
  put the selectedLine  into li
  put the selectedText  into tx
  journal empty
  journal "== la selection posee a la main =="
  journal "  selectedChunk : [" & ch & "]"
  journal "  selectedField : [" & fl & "]"
  journal "  selectedLine  : [" & li & "]"
  journal "  selectedText  : [" & tx & "]"
  journal "  attendu (mesure dans Basilisk, selection a la main) :"
  journal "    card field 1  et  line 1 of card field 1"
end relitSelection


-- ---- ce qui reste ouvert : le bouton « ouvert » --------------------
--
-- CES ESSAIS-LA SONT A PART, ET POUR UNE RAISON MESUREE : certains
-- LEVENT — « hide ch » le premier — et une erreur, dans l'application,
-- arrete le script sur un dialogue. Ils etaient d'abord dans le script
-- du bouton TORTURE, a la suite des autres : le rapport s'arretait sur
-- « hide ch » et l'on perdait la fin, le bilan compris. Le banc
-- s'arretait la ou il ne voulait QU'OBSERVER.
--
-- Deux precautions, donc :
--
--   1. leur propre bouton, pour que le rapport principal aille jusqu'au
--      bout sans un seul dialogue ;
--   2. UN GESTIONNAIRE PAR ESSAI RISQUE. Une erreur n'arrete que le
--      gestionnaire ou elle tombe : isoler chaque essai fait que le
--      suivant se joue quand meme. Sans cela, le premier refus emporte
--      les cinq autres et l'on ne mesure qu'une ligne sur six.
--
-- Aucune de ces lignes ne compte. Elles affichent ce que HC fait
-- aujourd'hui, pour qu'on le compare dans Basilisk : les inscrire en
-- reussites serait inscrire une supposition comme un fait.

on sectionOuvert
  global gCh, gCa
  titre "6. a mesurer dans HyperCard"

  -- Un morceau DANS un morceau.
  go to card "Atelier"
  put "alpha beta" & return & "gamma delta" into card field "A"
  note "le nombre de mots de la ligne 2", the number of words of line 2 of card field "A"
  note "le style d'un mot d'une ligne", the textStyle of word 1 of line 2 of card field "A"

  -- Une recherche qui ECHOUE remet-elle le curseur a zero ? HC le fait,
  -- parce que son curseur n'est rien d'autre que la derniere trouvaille.
  -- Ce qu'HyperCard en fait, personne ne l'a regarde.
  go to card "Atelier"
  put "alpha beta gamma delta" into bg field "T"
  find "alpha"
  find "zorglub"
  find "alpha"
  note "apres un echec, la recherche repart de", the short name of this card
  note "foundChunk apres un echec", the foundChunk

  -- LES SITES JUMEAUX DU DESIGNATEUR CALCULE.
  --
  -- « select the foundChunk » ne marchait pas ; c'est corrige, parce que
  -- l'idiome est dans le manuel et que le designateur n'existe que pour
  -- etre relu. Les QUATRE AUTRES commandes qui prennent un designateur
  -- ont ete mesurees du meme coup, et aucune n'est tranchee : HyperCard
  -- exige peut-etre « do "hide" && ch » pour toutes. Le banc est dans
  -- docs/mesures/designateur_calcule.txt.
  --
  -- gCh et gCa sont des GLOBALES, et non des variables locales : chaque
  -- essai vit dans son propre gestionnaire, et une locale n'y survivrait
  -- pas d'un appel a l'autre.
  go to card "Atelier"
  put "alpha beta" into card field "A"
  put "card field 1" into gCh
  put "card " & quote & "Deux" & quote into gCa

  essaiHide
  note "hide <variable> : le champ est-il encore visible", the visible of card field 1
  show card field 1

  essaiFindIn
  note "find in <variable>", the result

  essaiGoTo
  note "go to <variable> : carte atteinte", the short name of this card
  go to card "Atelier"

  essaiLire
end sectionOuvert

-- Les cinq essais, un par gestionnaire. Celui qui leve n'emporte que
-- lui-meme, et « note » s'ecrit dans sectionOuvert, APRES le retour.
on essaiHide
  global gCh
  hide gCh
end essaiHide

on essaiFindIn
  global gCh
  find "bet" in gCh
end essaiFindIn

on essaiGoTo
  global gCa
  go to gCa
end essaiGoTo

on essaiLire
  global gCh
  note "lire une propriete sur <variable>", the name of gCh
end essaiLire
