-- ════════════════════════════════════════════════════════════════════
--  TORTURE 3 — LES EXPRESSIONS TORDUES — LE SCRIPT DU BOUTON « TORTURE »
--
--  Les essais ne sont PAS dans ce script : ils sont dans le champ
--  « Essais », une expression par ligne, sous la forme
--
--      expression ==> attendu
--
--  et l'on peut en AJOUTER a la main, sans toucher a rien d'autre.
--
--    attendu ordinaire   la ligne dit « ok » ou « ECHEC » (les deux
--                        valeurs suivent)
--    ?                   personne n'a mesure ce qu'HyperCard rend : la
--                        valeur s'affiche SANS verdict
--    ERREUR              l'expression DOIT lever une erreur
--
--  Une ligne qui commence par « -- » est un titre de section ; une ligne
--  vide est sautee.
--
--  LA COMPARAISON SE FAIT AVEC « is », donc sans tenir compte des
--  majuscules et en comparant les nombres comme des nombres : 14 et
--  14.0 sont egaux, « B » et « b » aussi.
-- ════════════════════════════════════════════════════════════════════

on mouseUp
  razCompteurs
  set the lockErrorDialogs to true
  put card field "Essais" of card "Atelier" into lesEssais
  repeat with i = 1 to the number of lines of lesEssais
    put line i of lesEssais into l
    if l is empty then next repeat
    if char 1 to 2 of l is "--" then
      titre char 3 to length(l) of l
      next repeat
    end if
    put offset(" ==> ", l) into p
    if p is 0 then
      journal "  ???    ligne " & i & " sans « ==> » : " & l
      next repeat
    end if
    essai char 1 to p - 1 of l, char p + 5 to length(l) of l
  end repeat
  bilan
end mouseUp

-- Un essai, et son verdict.
on essai e, attendu
  global gR, gFini
  put false into gFini
  put empty into gR
  calcule e
  if gFini is true then put gR into obtenu
  else put "ERREUR" into obtenu
  if attendu is "?" then note e, obtenu
  else egal e & "  ==>  " & attendu, obtenu, attendu
end essai

-- LE CALCUL VIT DANS SON PROPRE GESTIONNAIRE : une erreur arrete
-- celui-ci, pas la boucle. « gFini » ne passe a true que si la ligne
-- d'avant est allee au bout.
on calcule e
  global gR, gFini
  do "put " & e & " into gR"
  put true into gFini
end calcule
